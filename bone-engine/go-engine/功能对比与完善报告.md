## 7. 使用示例

### 7.1 Repository 使用

```go
package main

import (
    "context"
    "database/sql"
    "github.com/bone-engine/bone-metadata-go"
    "github.com/bone-engine/bone-metadata-go/sql/executor"
)

type User struct {
    ID       int64  `bone:"id,primary,autoincr"`
    Username string `bone:"username"`
    Email    string `bone:"email"`
    Deleted  bool   `bone:"deleted"`
}

func (u *User) GetID() interface{} {
    return u.ID
}

func main() {
    // 创建 DB 连接
    db, _ := sql.Open("mysql", "user:password@/db")
    cfg := &config.Config{ /* 配置 */ }
    exec := executor.NewDefaultExecutor(db, cfg)
    
    // 创建 Repository
    repo := bonemetadata.NewBaseRepository[User, int64](exec)
    ctx := context.Background()
    
    // 查询
    user, _ := repo.FindById(ctx, 1)
    users, _ := repo.FindByIds(ctx, []int64{1, 2, 3})
    
    // Criteria 查询
    crit := criteria.New()
    crit.Eq("username", "admin").Like("email", "%@example.com").Page(1, 10)
    results, _ := repo.FindByCriteria(ctx, crit)
    
    // 插入
    newUser := &User{Username: "test", Email: "test@example.com"}
    id, _ := repo.Insert(ctx, newUser)
    
    // 更新
    user.Email = "new@example.com"
    _, _ = repo.Update(ctx, user)
    
    // 删除
    repo.DeleteById(ctx, id)
}
```

### 7.2 扩展引擎使用

```go
package main

import (
    "context"
    "github.com/bone-engine/bone-extension-go/extension"
    "github.com/bone-engine/bone-extension-go/registry"
)

func main() {
    // 创建扩展
    ext := extension.New("my-ext", "user.create", func(ctx *extension.Context) error {
        // 扩展逻辑
        return nil
    }).WithName("My Extension").WithPriority(100)
    
    // 注册扩展
    reg := registry.New()
    reg.Register(ext)
    
    // 执行扩展
    ctx := extension.NewContext(context.Background())
    ext.Execute(ctx)
}
```
