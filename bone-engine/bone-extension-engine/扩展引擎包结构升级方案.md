### 3. **核心引擎（core）**

```java
// 扩展注册中心
@Component
public class ExtensionRegistry {
    private final Map<Class<?>, List<ExtensionDefinition>> extensionMap = new ConcurrentHashMap<>();
    private final ExtensionScanner scanner;
    
    @Autowired
    public ExtensionRegistry(ExtensionScanner scanner) {
        this.scanner = scanner;
    }
    
    @PostConstruct
    public void init() {
        // 扫描并注册所有扩展点
        Map<Class<?>, List<ExtensionDefinition>> scannedExtensions = scanner.scan();
        extensionMap.putAll(scannedExtensions);
    }
    
    public void registerExtension(Class<?> extensionInterface, ExtensionDefinition definition) {
        extensionMap.computeIfAbsent(extensionInterface, k -> new ArrayList<>())
                   .add(definition);
    }
    
    public List<ExtensionDefinition> getExtensions(Class<?> extensionInterface) {
        return extensionMap.getOrDefault(extensionInterface, new ArrayList<>());
    }
    
    public List<ExtensionDefinition> findExtensions(Class<?> extensionInterface, BizContext context) {
        return getExtensions(extensionInterface).stream()
                .filter(ext -> ext.isEnabled() && ext.matches(context))
                .collect(Collectors.toList());
    }
}
```

```java
// 默认路由器实现
@Component
public class DefaultExtensionRouter implements ExtensionRouter {
    private final ExtensionRegistry registry;
    private final List<RouteMatcher> matchers;
    private final RoutingCache cache;
    
    @Autowired
    public DefaultExtensionRouter(ExtensionRegistry registry, RoutingCache cache) {
        this.registry = registry;
        this.cache = cache;
        this.matchers = Arrays.asList(
            new TenantMatcher(),
            new BizCodeMatcher(),
            new ScenarioMatcher()
        );
    }
    
    @Override
    public ExtensionDefinition route(Class<?> extensionInterface, BizContext context) {
        String cacheKey = generateCacheKey(extensionInterface, context);
        
        // 先查缓存
        ExtensionDefinition cached = cache.get(cacheKey);
        if (cached != null) {
            return cached;
        }
        
        // 获取所有候选扩展
        List<ExtensionDefinition> candidates = registry.findExtensions(extensionInterface, context);
        
        if (candidates.isEmpty()) {
            throw new ExtensionNotFoundException("No extension found for: " + extensionInterface.getName());
        }
        
        // 执行路由选择
        ExtensionDefinition selected = selectBestMatch(candidates, context);
        
        // 缓存结果
        cache.put(cacheKey, selected);
        return selected;
    }
    
    private ExtensionDefinition selectBestMatch(List<ExtensionDefinition> candidates, BizContext context) {
        // 1. 按优先级排序
        candidates.sort(Comparator.comparing(ExtensionDefinition::getPriority));
        
        // 2. 权重选择
        return new WeightSelector().select(candidates, context);
    }
    
    private String generateCacheKey(Class<?> extensionInterface, BizContext context) {
        return extensionInterface.getName() + ":" + context.getTenantCode() + ":" + context.getBizCode();
    }
}
```

```java
// 默认执行器
@Component
public class DefaultExtensionExecutor implements ExtensionExecutor {
    private final List<ExtensionInterceptor> interceptors;
    
    @Autowired
    public DefaultExtensionExecutor(List<ExtensionInterceptor> interceptors) {
        this.interceptors = interceptors != null ? interceptors : new ArrayList<>();
    }
    
    @Override
    public Object execute(ExtensionDefinition extension, Method method, Object[] args) {
        // 前置拦截
        for (ExtensionInterceptor interceptor : interceptors) {
            interceptor.beforeExecute(extension, method, args);
        }
        
        try {
            // 执行扩展方法
            Object result = method.invoke(extension.getInstance(), args);
            
            // 后置拦截
            for (ExtensionInterceptor interceptor : interceptors) {
                interceptor.afterExecute(extension, method, args, result);
            }
            
            return result;
        } catch (Exception e) {
            // 异常拦截
            for (ExtensionInterceptor interceptor : interceptors) {
                interceptor.onException(extension, method, args, e);
            }
            throw new ExtensionException("Extension execution failed", e);
        }
    }
}
```

```java
// 调用处理器
public class ExtensionInvocationHandler implements InvocationHandler {
    private final Class<?> extensionInterface;
    private final ExtensionRouter router;
    private final ExtensionExecutor executor;
    
    public ExtensionInvocationHandler(Class<?> extensionInterface, ExtensionRouter router, ExtensionExecutor executor) {
        this.extensionInterface = extensionInterface;
        this.router = router;
        this.executor = executor;
    }
    
    @Override
    public Object invoke(Object proxy, Method method, Object[] args) throws Throwable {
        if (isObjectMethod(method)) {
            return method.invoke(this, args);
        }
        
        // 查找BizContext参数
        BizContext<?> context = findBizContext(args);
        if (context == null) {
            throw new ExtensionException("BizContext parameter is required");
        }
        
        // 路由到具体的扩展实现
        ExtensionDefinition extension = router.route(extensionInterface, context);
        
        // 执行扩展方法
        return executor.execute(extension, method, args);
    }
    
    private boolean isObjectMethod(Method method) {
        return method.getDeclaringClass() == Object.class;
    }
    
    private BizContext<?> findBizContext(Object[] args) {
        if (args == null) return null;
        
        for (Object arg : args) {
            if (arg instanceof BizContext) {
                return (BizContext<?>) arg;
            }
        }
        return null;
    }
}
```

```java
// 路由责任链
package com.bone.engine.extension.core.router;

public class RoutingChain {
    private final List<RoutingStrategy> strategies;
    
    public RoutingResult route(RoutingContext context) {
        for (RoutingStrategy strategy : strategies) {
            RoutingResult result = strategy.execute(context);
            if (result.isMatched()) {
                return result;
            }
        }
        return RoutingResult.notFound();
    }
}
```
