// bone-engine/bone-workflow/src/main/java/com/bone/engine/workflow/rule/RuleEngine.java
@Component
public class RuleEngine {
    
    private final RuleRepository ruleRepository;
    private final RuleParser ruleParser;
    private final RuleExecutor ruleExecutor;
    
    public RuleEngine(RuleRepository ruleRepository, RuleParser ruleParser, RuleExecutor ruleExecutor) {
        this.ruleRepository = ruleRepository;
        this.ruleParser = ruleParser;
        this.ruleExecutor = ruleExecutor;
    }
    
    /**
     * 执行业务规则
     */
    public <T> RuleExecutionResult executeRules(T entity, String operation) {
        List<BusinessRule> rules = ruleRepository.findByEntityAndOperation(
            entity.getClass().getSimpleName(), operation);
        
        RuleExecutionResult result = new RuleExecutionResult();
        
        for (BusinessRule rule : rules) {
            if (evaluateConditions(rule.getConditions(), entity)) {
                RuleExecutionResult ruleResult = ruleExecutor.executeActions(rule.getActions(), entity);
                result.merge(ruleResult);
                
                // 如果规则执行失败，根据配置决定是否继续执行后续规则
                if (!ruleResult.isSuccess() && rule.isStopOnFailure()) {
                    break;
                }
            }
        }
        
        return result;
    }
    
    private <T> boolean evaluateConditions(List<RuleCondition> conditions, T entity) {
        for (RuleCondition condition : conditions) {
            if (!evaluateCondition(condition, entity)) {
                return false;
            }
        }
        return true;
    }
    
    private <T> boolean evaluateCondition(RuleCondition condition, T entity) {
        Object fieldValue = getFieldValue(entity, condition.getField());
        return ConditionEvaluator.evaluate(condition, fieldValue);
    }
    
    private Object getFieldValue(Object entity, String fieldName) {
        try {
            Field field = entity.getClass().getDeclaredField(fieldName);
            field.setAccessible(true);
            return field.get(entity);
        } catch (Exception e) {
            throw new RuleExecutionException("无法获取字段值: " + fieldName, e);
        }
    }
}

// 生成的代码 - PurchaseOrderService.java
@Service
@Transactional
public class PurchaseOrderServiceImpl implements PurchaseOrderService {
    
    private final PurchaseOrderRepository purchaseOrderRepository;
    private final RuleEngine ruleEngine;
    
    public PurchaseOrderServiceImpl(PurchaseOrderRepository purchaseOrderRepository, 
                                   RuleEngine ruleEngine) {
        this.purchaseOrderRepository = purchaseOrderRepository;
        this.ruleEngine = ruleEngine;
    }
    
    @Override
    public PurchaseOrder create(PurchaseOrder purchaseOrder) {
        // 执行业务规则验证
        RuleExecutionResult ruleResult = ruleEngine.executeRules(purchaseOrder, "CREATE");
        if (!ruleResult.isSuccess()) {
            throw new BusinessException("RULE_VALIDATION_FAILED", ruleResult.getMessages());
        }
        
        // 生成订单编号
        if (purchaseOrder.getOrderNumber() == null) {
            purchaseOrder.setOrderNumber(generateOrderNumber());
        }
        
        // 设置默认值
        if (purchaseOrder.getOrderDate() == null) {
            purchaseOrder.setOrderDate(LocalDateTime.now());
        }
        if (purchaseOrder.getCurrency() == null) {
            purchaseOrder.setCurrency("CNY");
        }
        if (purchaseOrder.getOrderStatus() == null) {
            purchaseOrder.setOrderStatus("DRAFT");
        }
        if (purchaseOrder.getTotalAmount() == null) {
            purchaseOrder.setTotalAmount(BigDecimal.ZERO);
        }
        
        return purchaseOrderRepository.save(purchaseOrder);
    }
    
    private String generateOrderNumber() {
        // 实现订单编号生成逻辑
        return "PO-" + LocalDate.now().format(DateTimeFormatter.BASIC_ISO_DATE) + "-" + 
               String.format("%06d", SequenceGenerator.next("purchase_order"));
    }
}
