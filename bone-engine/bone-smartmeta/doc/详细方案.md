#### 2. 规则引擎实现

```java
// bone-engine/bone-workflow/src/main/java/com/bone/engine/workflow/rule/RuleEngine.java
@Component
import lombok.extern.slf4j.Slf4j;
import org.springframework.util.ReflectionUtils;
import java.util.List;
import java.util.Map;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CompletableFuture;
import java.util.stream.Collectors;

@Slf4j
@Component
public class RuleEngine {
    
    private final RuleRepository ruleRepository;
    private final RuleParser ruleParser;
    private final RuleExecutor ruleExecutor;
    private final ExpressionEvaluator expressionEvaluator;
    
    // 缓存规则评估结果，提高性能
    private final Map<String, List<BusinessRule>> ruleCache = new ConcurrentHashMap<>();
    
    /**
     * 构造函数
     * 
     * @param ruleRepository 规则仓库
     * @param ruleParser 规则解析器
     * @param ruleExecutor 规则执行器
     * @param expressionEvaluator 表达式求值器
     */
    public RuleEngine(RuleRepository ruleRepository, 
                      RuleParser ruleParser, 
                      RuleExecutor ruleExecutor,
                      ExpressionEvaluator expressionEvaluator) {
        this.ruleRepository = Objects.requireNonNull(ruleRepository, "RuleRepository cannot be null");
        this.ruleParser = Objects.requireNonNull(ruleParser, "RuleParser cannot be null");
        this.ruleExecutor = Objects.requireNonNull(ruleExecutor, "RuleExecutor cannot be null");
        this.expressionEvaluator = Objects.requireNonNull(expressionEvaluator, "ExpressionEvaluator cannot be null");
        log.debug("RuleEngine initialized with dependencies");
    }
    
    /**
     * 执行业务规则
     * 
     * @param entity 业务实体
     * @param operation 操作类型
     * @param <T> 实体类型
     * @return 规则执行结果
     * @throws RuleExecutionException 当规则执行失败时抛出
     */
    public <T> RuleExecutionResult executeRules(T entity, String operation) {
        Objects.requireNonNull(entity, "Entity cannot be null");
        Objects.requireNonNull(operation, "Operation cannot be null or empty");
        
        String entityName = entity.getClass().getSimpleName();
        log.info("Executing rules for entity: {}, operation: {}", entityName, operation);
        
        try {
            List<BusinessRule> rules = getRules(entityName, operation);
            log.debug("Found {} rules to evaluate", rules.size());
            
            RuleExecutionResult result = new RuleExecutionResult();
            
            for (BusinessRule rule : rules) {
                if (evaluateConditions(rule.getConditions(), entity)) {
                    log.debug("Executing rule: {}", rule.getName());
                    RuleExecutionResult ruleResult = ruleExecutor.executeActions(rule.getActions(), entity);
                    result.merge(ruleResult);
                    
                    // 如果规则执行失败，根据配置决定是否继续执行后续规则
                    if (!ruleResult.isSuccess() && rule.isStopOnFailure()) {
                        log.info("Stopping rule execution as rule {} failed with stopOnFailure=true", rule.getName());
                        break;
                    }
                } else {
                    log.debug("Skipping rule: {} as conditions not met", rule.getName());
                }
            }
            
            log.debug("Rule execution completed with result: {}", result.isSuccess());
            return result;
        } catch (Exception e) {
            log.error("Failed to execute rules for entity: {}, operation: {}", entityName, operation, e);
            throw new RuleExecutionException("规则执行失败: " + e.getMessage(), e);
        }
    }
    
    /**
     * 异步执行规则
     * 
     * @param entity 业务实体
     * @param operation 操作类型
     * @param <T> 实体类型
     * @return 包含规则执行结果的CompletableFuture
     */
    public <T> CompletableFuture<RuleExecutionResult> executeRulesAsync(T entity, String operation) {
        return CompletableFuture.supplyAsync(() -> executeRules(entity, operation))
            .exceptionally(ex -> {
                log.error("Async rule execution failed", ex);
                RuleExecutionResult result = new RuleExecutionResult();
                result.setSuccess(false);
                result.addMessage("异步规则执行失败: " + ex.getMessage());
                return result;
            });
    }
    
    /**
     * 批量执行规则
     * 
     * @param entities 业务实体列表
     * @param operation 操作类型
     * @param <T> 实体类型
     * @return 批量执行结果映射
     */
    public <T> Map<T, RuleExecutionResult> executeRulesBatch(List<T> entities, String operation) {
        return entities.stream()
            .collect(Collectors.toMap(
                entity -> entity,
                entity -> executeRules(entity, operation)
            ));
    }
    
    /**
     * 评估规则条件
     * 
     * @param conditions 规则条件列表
     * @param entity 业务实体
     * @param <T> 实体类型
     * @return 所有条件都满足返回true，否则返回false
     */
    private <T> boolean evaluateConditions(List<RuleCondition> conditions, T entity) {
        if (conditions == null || conditions.isEmpty()) {
            return true; // 没有条件则默认通过
        }
        
        return conditions.stream()
            .allMatch(condition -> evaluateCondition(condition, entity));
    }
    
    /**
     * 评估单个条件
     * 
     * @param condition 规则条件
     * @param entity 业务实体
     * @param <T> 实体类型
     * @return 条件满足返回true，否则返回false
     */
    private <T> boolean evaluateCondition(RuleCondition condition, T entity) {
        try {
            // 使用表达式求值器代替直接反射，支持更复杂的表达式
            if (condition.hasExpression()) {
                return expressionEvaluator.evaluate(condition.getExpression(), entity);
            } else {
                // 兼容旧版字段直接比较
                Object fieldValue = getFieldValue(entity, condition.getField());
                return ConditionEvaluator.evaluate(condition, fieldValue);
            }
        } catch (Exception e) {
            log.error("Failed to evaluate condition: {}", condition, e);
            return false; // 条件评估失败默认返回false
        }
    }
    
    /**
     * 获取字段值
     * 
     * @param entity 业务实体
     * @param fieldName 字段名
     * @return 字段值
     */
    private Object getFieldValue(Object entity, String fieldName) {
        try {
            // 使用Spring的ReflectionUtils更安全地访问字段
            Field field = ReflectionUtils.findField(entity.getClass(), fieldName);
            if (field == null) {
                // 尝试使用getter方法
                String getterMethodName = "get" + Character.toUpperCase(fieldName.charAt(0)) + fieldName.substring(1);
                return ReflectionUtils.invokeMethod(entity.getClass().getMethod(getterMethodName), entity);
            }
            
            ReflectionUtils.makeAccessible(field);
            return ReflectionUtils.getField(field, entity);
        } catch (Exception e) {
            log.warn("Could not get field value for: {}", fieldName, e);
            throw new RuleExecutionException("无法获取字段值: " + fieldName, e);
        }
    }
    
    /**
     * 获取规则列表（带缓存）
     * 
     * @param entityName 实体名称
     * @param operation 操作类型
     * @return 规则列表
     */
    private List<BusinessRule> getRules(String entityName, String operation) {
        String cacheKey = entityName + ":" + operation;
        
        return ruleCache.computeIfAbsent(cacheKey, key -> {
            log.debug("Cache miss for rules: {}", key);
            return ruleRepository.findByEntityAndOperation(entityName, operation);
        });
    }
    
    /**
     * 刷新规则缓存
     */
    public void refreshRuleCache() {
        ruleCache.clear();
        log.info("Rule cache refreshed");
    }
    
    /**
     * 刷新特定实体和操作的规则缓存
     * 
     * @param entityName 实体名称
     * @param operation 操作类型
     */
    public void refreshRuleCache(String entityName, String operation) {
        String cacheKey = entityName + ":" + operation;
        ruleCache.remove(cacheKey);
        log.debug("Rule cache refreshed for: {}", cacheKey);
    }
}
```

```java
// 生成的代码 - PurchaseOrderService.java
@Service
@Transactional
public class PurchaseOrderServiceImpl implements PurchaseOrderService {
    
    private final PurchaseOrderRepository purchaseOrderRepository;
    private final RuleEngine ruleEngine;
    
    public PurchaseOrderServiceImpl(PurchaseOrderRepository purchaseOrderRepository, 
                                   RuleEngine ruleEngine) {
        this.purchaseOrderRepository = purchaseOrderRepository;
        this.ruleEngine = ruleEngine;
    }
    
    @Override
    public PurchaseOrder create(PurchaseOrder purchaseOrder) {
        // 执行业务规则验证
        RuleExecutionResult ruleResult = ruleEngine.executeRules(purchaseOrder, "CREATE");
        if (!ruleResult.isSuccess()) {
            throw new BusinessException("RULE_VALIDATION_FAILED", ruleResult.getMessages());
        }
        
        // 生成订单编号
        if (purchaseOrder.getOrderNumber() == null) {
            purchaseOrder.setOrderNumber(generateOrderNumber());
        }
        
        // 设置默认值
        if (purchaseOrder.getOrderDate() == null) {
            purchaseOrder.setOrderDate(LocalDateTime.now());
        }
        if (purchaseOrder.getCurrency() == null) {
            purchaseOrder.setCurrency("CNY");
        }
        if (purchaseOrder.getOrderStatus() == null) {
            purchaseOrder.setOrderStatus("DRAFT");
        }
        if (purchaseOrder.getTotalAmount() == null) {
            purchaseOrder.setTotalAmount(BigDecimal.ZERO);
        }
        
        return purchaseOrderRepository.save(purchaseOrder);
    }
    
    private String generateOrderNumber() {
        // 实现订单编号生成逻辑
        return "PO-" + LocalDate.now().format(DateTimeFormatter.BASIC_ISO_DATE) + "-" + 
               String.format("%06d", SequenceGenerator.next("purchase_order"));
    }
}
```
