3.1 数据源路由工具（编程式切换 + 分片上下文）
/**
 * 数据源路由工具类（封装后业务直接用，避免重复代码）
 */
@Component
public class DataSourceRouter {
    private static final Logger log = LoggerFactory.getLogger(DataSourceRouter.class);

    /**
     * 1. 切换到指定数据源执行操作（含自动清理上下文）
     * @param dataSource 数据源名（如master、slave、tenant_001）
     * @param action 业务操作
     * @return 操作结果
     */
    public static <T> T execute(String dataSource, Supplier<T> action) {
        String preDs = DynamicDataSourceContextHolder.peek(); // 记录切换前的数据源
        try {
            log.debug("数据源切换：{} → {}", preDs, dataSource);
            DynamicDataSourceContextHolder.push(dataSource);
            return action.get();
        } finally {
            DynamicDataSourceContextHolder.poll(); // 必须清理，避免线程复用错乱
            log.debug("数据源恢复：{}", preDs);
        }
    }

    /**
     * 2. 无返回值操作（重载方法）
     */
    public static void execute(String dataSource, Runnable action) {
        execute(dataSource, () -> {
            action.run();
            return null;
        });
    }

    /**
     * 3. 分库分表场景：带分片键的操作（自动绑定分片上下文）
     */
    public static <T> T executeSharding(String shardDs, String shardCol, Object shardVal, Supplier<T> action) {
        return execute(shardDs, () -> {
            ShardingContextHolder.setShardingValue(shardCol, shardVal); // 绑定分片键
            try {
                return action.get();
            } finally {
                ShardingContextHolder.clear(); // 清理分片上下文
            }
        });
    }
}

/**
 * 分片上下文管理器（ThreadLocal存储分片键，避免业务传递）
 */
public class ShardingContextHolder {
    private static final ThreadLocal<Map<String, Object>> SHARD_CONTEXT = ThreadLocal.withInitial(HashMap::new);

    public static void setShardingValue(String column, Object value) {
        SHARD_CONTEXT.get().put(column, value);
    }

    public static Object getShardingValue(String column) {
        return SHARD_CONTEXT.get().get(column);
    }

    public static void clear() {
        SHARD_CONTEXT.remove(); // 必须清理，防内存泄漏
    }
}

3.2 多租户拦截器（自动解析租户 ID，零业务侵入）
/**
 * 多租户数据源拦截器（从请求头自动获取租户ID，切换数据源）
 */
@Component
public class TenantDataSourceInterceptor implements HandlerInterceptor {
    private static final String TENANT_HEADER = "X-Tenant-ID"; // 租户ID请求头
    private static final Logger log = LoggerFactory.getLogger(TenantDataSourceInterceptor.class);

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        // 1. 从请求头获取租户ID
        String tenantId = request.getHeader(TENANT_HEADER);
        if (StringUtils.isBlank(tenantId)) {
            throw new BizException("400", "租户ID（X-Tenant-ID）不能为空");
        }

        // 2. 拼接租户数据源名（与配置中的tenant_xxx对齐）
        String tenantDs = "tenant_" + tenantId;
        // 3. 检查数据源是否存在（避免非法租户）
        if (!DynamicDataSourceContextHolder.containsDataSource(tenantDs)) {
            log.warn("租户数据源不存在：{}", tenantDs);
            throw new BizException("404", "租户数据源未配置：" + tenantId);
        }

        // 4. 切换租户数据源
        DynamicDataSourceContextHolder.push(tenantDs);
        log.debug("多租户数据源切换：{}", tenantDs);
        return true;
    }

    // 请求结束后必须清理上下文
    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response, Object handler, Exception ex) {
        DynamicDataSourceContextHolder.clear();
    }
}

3.3 数据源健康检查（生产级监控）
/**
 * 自定义数据源健康检查（覆盖所有数据源，含主从、租户、分片）
 */
@Component
public class MultiDataSourceHealthIndicator implements HealthIndicator {
    @Autowired
    private DynamicRoutingDataSource dynamicDataSource;
    @Autowired(required = false)
    private DataSource shardUserDataSource; // 分片用户库
    @Autowired(required = false)
    private DataSource shardOrderDataSource; // 分片订单库

    @Override
    public Health health() {
        Map<String, Object> details = new LinkedHashMap<>();
        boolean allHealthy = true;

        // 1. 检查动态数据源（主从+多租户）
        Map<String, DataSource> dynamicDsMap = dynamicDataSource.getDataSources();
        for (Map.Entry<String, DataSource> entry : dynamicDsMap.entrySet()) {
            allHealthy &= checkSingleDataSource(entry.getKey(), entry.getValue(), details);
        }

        // 2. 检查分片数据源
        allHealthy &= checkSingleDataSource("shard_user", shardUserDataSource, details);
        allHealthy &= checkSingleDataSource("shard_order", shardOrderDataSource, details);

        // 3. 组装健康状态
        Health.Builder builder = allHealthy ? Health.up() : Health.down();
        return builder.withDetails(details).build();
    }

    /**
     * 检查单个数据源连通性（执行SELECT 1测试）
     */
    private boolean checkSingleDataSource(String dsName, DataSource ds, Map<String, Object> details) {
        if (ds == null) {
            details.put(dsName, "DOWN（数据源未初始化）");
            return false;
        }

        try (Connection conn = ds.getConnection(); Statement stmt = conn.createStatement()) {
            long start = System.currentTimeMillis();
            boolean valid = stmt.execute("SELECT 1");
            long cost = System.currentTimeMillis() - start;
            details.put(dsName, String.format("UP（响应时间：%dms）", cost));
            return valid;
        } catch (SQLException e) {
            details.put(dsName, String.format("DOWN（原因：%s）", e.getMessage()));
            return false;
        }
    }
}

    }
}

@Service
public class OrderServiceImpl implements OrderService {
    @Autowired
    private OrderMapper orderMapper;
    @Autowired
    private OrderItemMapper orderItemMapper;

    // 1. 创建订单：按user_id自动分片到t_order_0~7（无需手动指定表）
    @Override
    public void createOrder(Order order, List<OrderItem> items) {
        // 调用工具类，绑定分片键（user_id）
        DataSourceRouter.executeSharding("shard_order", "user_id", order.getUserId(), () -> {
            orderMapper.insert(order); // 自动路由到t_order_{user_id%8}
            // 订单项按order_id分片（与订单表对齐）
            items.forEach(item -> {
                item.setOrderId(order.getOrderId());
                orderItemMapper.insert(item); // 自动路由到t_order_item_{order_id%8}
            });
            return null;
        });
    }

    // 2. 查询用户订单：自动路由到对应分片表
    @Override
    public List<Order> getUserOrders(Long userId) {
        return DataSourceRouter.executeSharding("shard_order", "user_id", userId, () -> {
            return orderMapper.selectByUserId(userId); // 只查当前用户的分片表
        });
    }
}

5.1 动态数据源刷新（无需重启服务）
/**
 * Nacos配置监听（动态新增/修改数据源，适合多租户动态扩展）
 */
@Component
public class NacosDataSourceRefreshListener {
    @Autowired
    private DynamicRoutingDataSource dynamicDataSource;
    @Value("${spring.application.name}")
    private String appName;

    // 监听Nacos中数据源配置变更（dataId：{appName}-dynamic-ds.yaml）
    @NacosConfigListener(dataId = "${spring.application.name}-dynamic-ds.yaml", groupId = "PROD_GROUP")
    public void refreshDataSource(String newConfig) {
        try {
            // 1. 解析新配置（YAML→数据源Map）
            Yaml yaml = new Yaml();
            Map<String, Map<String, String>> configMap = yaml.loadAs(newConfig, Map.class);
            Map<String, DataSource> newDsMap = parseDataSourceConfig(configMap.get("datasources"));

            // 2. 新增/更新数据源（dynamic-datasource原生方法）
            dynamicDataSource.addDataSource(newDsMap);
            log.info("动态数据源刷新成功，当前数据源：{}", dynamicDataSource.getDataSources().keySet());
        } catch (Exception e) {
            log.error("动态数据源刷新失败", e);
            // 生产环境可发告警：钉钉/企业微信通知运维
            AlarmUtil.sendDingTalkAlarm("动态数据源刷新失败：" + e.getMessage());
        }
    }

    /**
     * 解析配置为Druid数据源（生产推荐用Druid，支持监控）
     */
    private Map<String, DataSource> parseDataSourceConfig(Map<String, String> dsConfigMap) {
        Map<String, DataSource> dsMap = new HashMap<>();
        for (Map.Entry<String, String> entry : dsConfigMap.entrySet()) {
            String dsName = entry.getKey();
            Map<String, String> config = entry.getValue();

            DruidDataSource druidDs = new DruidDataSource();
            druidDs.setUrl(config.get("url"));
            druidDs.setUsername(config.get("username"));
            druidDs.setPassword(config.get("password"));
            druidDs.setDriverClassName("com.mysql.cj.jdbc.Driver");
            // 连接池参数（与主配置对齐）
            druidDs.setMaxActive(20);
            druidDs.setMinIdle(3);
            druidDs.setMaxWait(30000);

            dsMap.put(dsName, druidDs);
        }
        return dsMap;
    }
}

5.2 避坑指南（生产必看）
坑点
原因分析
解决方案
数据源切换后事务无效
AOP 切面优先级低于事务切面，事务先开启
框架已默认设置@Order(Ordered.HIGHEST_PRECEDENCE)，无需手动改
线程复用导致数据源错乱
ThreadLocal 未清理，线程池复用上下文
所有切换操作必须在finally中调用DynamicDataSourceContextHolder.poll()
主从延迟导致读不到数据
写主库后立即读从库，数据未同步
关键场景（如支付后查订单）用@DS("master")强制读主库
分片表关联查询性能差
跨分片查询未对齐分片键
订单 / 订单项按同一键分片（如 order_id），避免跨表扫描
多租户数据源不存在
非法租户 ID 或配置未同步
拦截器中检查数据源存在性，不存在直接抛 404，避免服务报错

六、方案优势总结
极简侵入：业务代码仅用@DS注解或工具类，无需改 Mapper/DAO，接入成本低；
全场景覆盖：读写分离、多租户、分库分表、跨库事务一次搞定；
生产就绪：含配置加密、健康检查、动态刷新、故障转移，满足企业级需求；
性能优化：连接池参数精细化配置，分片查询避免跨表扫描，读写分离减轻主库压力；
易运维：监控接口暴露数据源状态，动态刷新无需重启，问题可追溯。
